struct Node {
    int value;
    struct Node* next;
};

typedef struct Node Node;

Node* reverse_list(Node* head) {
    Node* previous = null;
    Node* current = head;

    while (current != null) {
        Node* next = current->next;
        current->next = previous;
        previous = current;
        current = next;
    }

    return previous;
}

int main() {

    first->value = 1;
    first->next = second;
    second->value = 2;
    second->next = third;
    third->value = 3;
    third->next = null;

    Node* reversed = reverse_list(first);

    if (reversed->value != 3) {
        return 1;
    }
    if (reversed->next->value != 2) {
        return 1;
    }
    if (reversed->next->next->value != 1) {
        return 1;
    }
    if (reversed->next->next->next != null) {
        return 1;
    }

    return 9;
}
